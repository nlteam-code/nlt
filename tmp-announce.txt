function esc(s: string) { return s.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;"); }
function html(t: string) {
  const b = esc(t)
    .replace("app.newleveltraining.bg", '<a href="https://app.newleveltraining.bg" style="color:#e30613;font-weight:bold">app.newleveltraining.bg</a>')
    .replace(/^(Какво можеш да правиш още сега:|Какво предстои:)$/gm, "<b>$1</b>")
    .replace(/^• ([^–]+) –/gm, "• <b>$1</b> –")
    .replace(/\n/g, "<br>");
  return `<div style="font-family:Arial,Helvetica,sans-serif;font-size:15px;line-height:1.55;color:#111;max-width:600px">${b}</div>`;
}
const okEmail = (e: string | null) => { const s = (e || "").trim(); return s.indexOf("@") > 0 && s.includes(".") && !/\s/.test(s); };

Deno.serve(async (req) => {
  if (req.headers.get("x-cron-secret") !== Deno.env.get("CRON_SECRET")) return new Response("forbidden", { status: 403 });
  const { mode } = await req.json().catch(() => ({}));
  const key = Deno.env.get("RESEND_API_KEY")!;
  const sb = createClient(Deno.env.get("SUPABASE_URL")!, Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!);
  const msg = (to: string) => ({ from: FROM, to: [to], reply_to: REPLY_TO, subject: SUBJECT, text: TEXT, html: html(TEXT) });

  if (mode === "test") {
    const r = await fetch("https://api.resend.com/emails", { method: "POST", headers: { Authorization: `Bearer ${key}`, "Content-Type": "application/json" }, body: JSON.stringify(msg(OWNER)) });
    return new Response(JSON.stringify({ test: r.status, body: await r.text() }), { headers: { "Content-Type": "application/json" } });
  }
  if (mode !== "all") return new Response("mode?", { status: 400 });

  const { data: cl, error } = await sb.from("clients").select("id,email,status").in("status", ["Active", "Lead"]);
  if (error) return new Response(error.message, { status: 500 });
  const { data: done } = await sb.from("announce_log").select("client_id").eq("campaign", CAMPAIGN);
  const sent = new Set((done || []).map((d: any) => d.client_id));
  const seen = new Set<string>();
  const list = (cl || []).filter((c: any) => {
    if (!okEmail(c.email) || sent.has(c.id)) return false;
    const e = c.email.trim().toLowerCase(); if (seen.has(e)) return false; seen.add(e); return true;
  });
  if (list.length > MAX_ALL) return new Response(`too many: ${list.length}`, { status: 400 });

  let ok = 0; const errs: string[] = [];
  for (let i = 0; i < list.length; i += 50) {
    const chunk = list.slice(i, i + 50);
    const r = await fetch("https://api.resend.com/emails/batch", { method: "POST", headers: { Authorization: `Bearer ${key}`, "Content-Type": "application/json" }, body: JSON.stringify(chunk.map((c: any) => msg(c.email.trim()))) });
    if (!r.ok) { errs.push(`${r.status} ${await r.text()}`); continue; }
    const j = await r.json(); const ids = (j.data || []).map((d: any) => d.id);
    await sb.from("announce_log").insert(chunk.map((c: any, k: number) => ({ client_id: c.id, campaign: CAMPAIGN, resend_id: ids[k] || null })));
    ok += chunk.length;
    await new Promise((s) => setTimeout(s, 1200));
  }
  return new Response(JSON.stringify({ candidates: list.length, sent: ok, errors: errs }), { headers: { "Content-Type": "application/json" } });
});
